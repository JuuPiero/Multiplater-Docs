PHẦN II · LẬP TRÌNH SOCKET CHO GAME

Ví dụ thực hành: Game Server UDP tối giản

Chương này tổng hợp toàn bộ kiến thức của Phần II (chương 9-13) và tick loop cố định của Phần I (chương 4) thành một ví dụ hoàn chỉnh, biên dịch chạy được: một game server UDP nhận vị trí (x, y) từ nhiều client cùng lúc, lưu vào một bảng ánh xạ theo địa chỉ, và ở mỗi tick cố định (20 lần/giây) broadcast snapshot vị trí của mọi người chơi cho toàn bộ client đang kết nối. Cả server và client đều biên dịch chạy được trên Windows (Winsock2) lẫn Linux (Berkeley sockets) qua #ifdef _WIN32.

1. Kiến trúc tổng thể của ví dụ

sequenceDiagram participant CA as Client A participant CB as Client B participant S as Server (1 UDP socket, non-blocking) CA->>S: PacketType::Input (x, y gia lap di chuyen) S->>S: Cap nhat PlayerInfo cho A (tao moi neu chua tung thay) CB->>S: PacketType::Input (x, y gia lap di chuyen) S->>S: Cap nhat PlayerInfo cho B Note over S: Moi tick 50ms (20Hz) S->>CA: PacketType::Snapshot (vi tri TAT CA nguoi choi) S->>CB: PacketType::Snapshot (vi tri TAT CA nguoi choi) Note over S: Quet dinh ky: client nao qua TIMEOUT_SECONDS khong gui gi -> loai bo

Các thành phần tái sử dụng trực tiếp từ các chương trước:

Thành phầnNguồn
Non-blocking socket + vòng lặp đọc hết gói đang chờChương 9
Packet header (magic, type, sequence, payload length) + serialize thủ côngChương 10
Bảng ánh xạ địa chỉ → người chơi (AddrKey/AddrKeyHash)Chương 9
Phát hiện timeout, dọn dẹp client chếtChương 12
Vòng lặp tick cố định (fixed tick rate)Chương 4, Phần I
Đơn giản hóa có chủ đích: Để giữ ví dụ này gọn và tập trung vào luồng dữ liệu cốt lõi, phần reliable UDP (chương 11) và tách đa luồng (chương 13) không được đưa vào — dữ liệu vị trí ở đây gửi qua kênh Unreliable đơn giản (đúng bản chất dữ liệu vị trí liên tục, xem bảng kênh ở chương 11), và toàn bộ chạy đơn luồng cho dễ đọc. Phần bài tập mở rộng cuối chương gợi ý cách bổ sung các phần này.

2. Mã nguồn dùng chung: Protocol.h

File header dùng chung giữa cả server và client, gộp lại các định nghĩa packet đã học ở chương 10.

C++ — Protocol.h
#pragma once
#ifdef _WIN32
  #include <winsock2.h>
  #include <ws2tcpip.h>
  #pragma comment(lib, "Ws2_32.lib")
  using socket_t = SOCKET;
  constexpr socket_t INVALID_SOCK = INVALID_SOCKET;
#else
  #include <sys/socket.h>
  #include <netinet/in.h>
  #include <arpa/inet.h>
  #include <unistd.h>
  #include <fcntl.h>
  #include <cerrno>
  using socket_t = int;
  constexpr socket_t INVALID_SOCK = -1;
#endif
#include <cstdint>
#include <cstring>
#include <cstdio>

constexpr uint32_t PROTOCOL_MAGIC = 0x47414D45;
constexpr int MAX_PACKET_SIZE = 1200;
constexpr int MAX_PLAYERS_PER_SNAPSHOT = 32;

enum class PacketType : uint8_t {
    Input      = 0,
    Snapshot   = 1,
    Disconnect = 2,
};

struct PacketHeader {
    uint32_t magic;
    PacketType type;
    uint16_t sequence;
    uint16_t payloadLength;
};

struct PacketWriter {
    uint8_t* buffer; size_t capacity; size_t offset = 0;
    bool WriteU8(uint8_t v) { if (offset+1>capacity) return false; buffer[offset++]=v; return true; }
    bool WriteU16(uint16_t v) { if (offset+2>capacity) return false; uint16_t n=htons(v); memcpy(buffer+offset,&n,2); offset+=2; return true; }
    bool WriteU32(uint32_t v) { if (offset+4>capacity) return false; uint32_t n=htonl(v); memcpy(buffer+offset,&n,4); offset+=4; return true; }
    bool WriteFloat(float v) { uint32_t bits; memcpy(&bits,&v,4); return WriteU32(bits); }
};

struct PacketReader {
    const uint8_t* buffer; size_t size; size_t offset = 0;
    bool ReadU8(uint8_t& o) { if (offset+1>size) return false; o=buffer[offset++]; return true; }
    bool ReadU16(uint16_t& o) { if (offset+2>size) return false; uint16_t n; memcpy(&n,buffer+offset,2); o=ntohs(n); offset+=2; return true; }
    bool ReadU32(uint32_t& o) { if (offset+4>size) return false; uint32_t n; memcpy(&n,buffer+offset,4); o=ntohl(n); offset+=4; return true; }
    bool ReadFloat(float& o) { uint32_t bits; if(!ReadU32(bits)) return false; memcpy(&o,&bits,4); return true; }
};

inline bool WriteHeader(PacketWriter& w, PacketType type, uint16_t seq, uint16_t len) {
    return w.WriteU32(PROTOCOL_MAGIC) && w.WriteU8((uint8_t)type)
        && w.WriteU16(seq) && w.WriteU16(len);
}

inline bool ReadHeader(PacketReader& r, PacketHeader& h) {
    uint8_t rawType;
    if (!r.ReadU32(h.magic) || h.magic != PROTOCOL_MAGIC) return false;
    if (!r.ReadU8(rawType)) return false;
    h.type = (PacketType)rawType;
    return r.ReadU16(h.sequence) && r.ReadU16(h.payloadLength);
}

inline bool SetSocketNonBlocking(socket_t sock) {
#ifdef _WIN32
    u_long mode = 1;
    return ioctlsocket(sock, FIONBIO, &mode) == 0;
#else
    int flags = fcntl(sock, F_GETFL, 0);
    if (flags == -1) return false;
    return fcntl(sock, F_SETFL, flags | O_NONBLOCK) != -1;
#endif
}

inline bool WasWouldBlock() {
#ifdef _WIN32
    return WSAGetLastError() == WSAEWOULDBLOCK;
#else
    return errno == EWOULDBLOCK || errno == EAGAIN;
#endif
}

inline void PlatformStartup() {
#ifdef _WIN32
    WSADATA wsaData;
    WSAStartup(MAKEWORD(2, 2), &wsaData);
#endif
}

inline void PlatformShutdown() {
#ifdef _WIN32
    WSACleanup();
#endif
}

inline void CloseSock(socket_t s) {
#ifdef _WIN32
    closesocket(s);
#else
    close(s);
#endif
}

3. Server hoàn chỉnh: server_main.cpp

C++ — server_main.cpp
// Bien dich Windows (MinGW-w64): g++ server_main.cpp -lws2_32 -o server.exe
// Bien dich Linux:                g++ server_main.cpp -o server
#include "Protocol.h"
#include <unordered_map>
#include <chrono>
#include <thread>

constexpr uint16_t SERVER_PORT = 9000;
constexpr double TICK_INTERVAL = 1.0 / 20.0;   // 20 Hz
constexpr double TIMEOUT_SECONDS = 10.0;

struct AddrKey {
    uint32_t ip; uint16_t port;
    bool operator==(const AddrKey& o) const { return ip==o.ip && port==o.port; }
};
struct AddrKeyHash {
    size_t operator()(const AddrKey& k) const { return (size_t(k.ip)<<16) ^ k.port; }
};
AddrKey MakeKey(const sockaddr_in& a) { return AddrKey{ a.sin_addr.s_addr, a.sin_port }; }

struct PlayerInfo {
    int id; sockaddr_in addr; float x=0, y=0; double lastReceivedTime;
};

std::unordered_map<AddrKey, PlayerInfo, AddrKeyHash> g_players;
int g_nextId = 1;

double NowSeconds() {
    using namespace std::chrono;
    return duration<double>(steady_clock::now().time_since_epoch()).count();
}

void HandleInputPacket(const uint8_t* payload, size_t len, const sockaddr_in& from) {
    PacketReader r{ payload, len };
    float x, y;
    if (!r.ReadFloat(x) || !r.ReadFloat(y)) return;

    AddrKey key = MakeKey(from);
    auto it = g_players.find(key);
    if (it == g_players.end()) {
        PlayerInfo p{}; p.id = g_nextId++; p.addr = from;
        printf("Player moi tham gia: id=%d\n", p.id);
        it = g_players.emplace(key, p).first;
    }
    it->second.x = x;
    it->second.y = y;
    it->second.lastReceivedTime = NowSeconds();
}

void HandleDisconnectPacket(const sockaddr_in& from) {
    AddrKey key = MakeKey(from);
    auto it = g_players.find(key);
    if (it != g_players.end()) {
        printf("Player id=%d thoat game (graceful disconnect)\n", it->second.id);
        g_players.erase(it);
    }
}

void ProcessAllPendingPackets(socket_t sock) {
    uint8_t buf[MAX_PACKET_SIZE];
    for (;;) {
        sockaddr_in from{};
#ifdef _WIN32
        int fromLen = sizeof(from);
#else
        socklen_t fromLen = sizeof(from);
#endif
        int n = recvfrom(sock, (char*)buf, sizeof(buf), 0, (sockaddr*)&from, &fromLen);
        if (n <= 0) {
            if (!WasWouldBlock()) { /* loi that su, co the log */ }
            break;
        }
        PacketReader r{ buf, (size_t)n };
        PacketHeader h;
        if (!ReadHeader(r, h)) continue;   // sai magic hoac goi hong - loai bo am tham

        if (h.type == PacketType::Input) {
            HandleInputPacket(buf + r.offset, h.payloadLength, from);
        } else if (h.type == PacketType::Disconnect) {
            HandleDisconnectPacket(from);
        }
    }
}

void CheckTimeouts() {
    double now = NowSeconds();
    for (auto it = g_players.begin(); it != g_players.end(); ) {
        if (now - it->second.lastReceivedTime > TIMEOUT_SECONDS) {
            printf("Player id=%d TIMEOUT - loai khoi server\n", it->second.id);
            it = g_players.erase(it);
        } else {
            ++it;
        }
    }
}

void BroadcastSnapshot(socket_t sock, uint16_t& snapshotSeq) {
    uint8_t buf[MAX_PACKET_SIZE];
    PacketWriter w{ buf, sizeof(buf) };

    int count = (int)g_players.size();
    if (count > MAX_PLAYERS_PER_SNAPSHOT) count = MAX_PLAYERS_PER_SNAPSHOT;

    // Header ghi tam, se cap nhat payloadLength sau khi biet chinh xac do dai
    WriteHeader(w, PacketType::Snapshot, snapshotSeq++, 0);
    w.WriteU8((uint8_t)count);
    int written = 0;
    for (auto& [key, p] : g_players) {
        if (written >= count) break;
        w.WriteU16((uint16_t)p.id);
        w.WriteFloat(p.x);
        w.WriteFloat(p.y);
        ++written;
    }

    // Gui cung 1 snapshot cho tat ca client dang ket noi (broadcast don gian)
    for (auto& [key, p] : g_players) {
        sendto(sock, (const char*)buf, (int)w.offset, 0,
               (const sockaddr*)&p.addr, sizeof(p.addr));
    }
}

int main() {
    PlatformStartup();

    socket_t sock = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
    if (sock == INVALID_SOCK) { fprintf(stderr, "socket() that bai\n"); return 1; }

    sockaddr_in addr{};
    addr.sin_family = AF_INET;
    addr.sin_port = htons(SERVER_PORT);
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    if (bind(sock, (sockaddr*)&addr, sizeof(addr)) != 0) {
        fprintf(stderr, "bind() that bai tren cong %d\n", SERVER_PORT); return 1;
    }
    if (!SetSocketNonBlocking(sock)) { fprintf(stderr, "khong the non-blocking\n"); return 1; }

    printf("Game server UDP dang chay tren cong %d (tick 20Hz)\n", SERVER_PORT);

    uint16_t snapshotSeq = 0;
    double lastTimeoutCheck = NowSeconds();

    while (true) {
        double frameStart = NowSeconds();

        ProcessAllPendingPackets(sock);

        if (frameStart - lastTimeoutCheck > 1.0) {   // quet timeout moi ~1 giay, khong can moi tick
            CheckTimeouts();
            lastTimeoutCheck = frameStart;
        }

        BroadcastSnapshot(sock, snapshotSeq);

        double elapsed = NowSeconds() - frameStart;
        double sleepTime = TICK_INTERVAL - elapsed;
        if (sleepTime > 0) {
            std::this_thread::sleep_for(std::chrono::duration<double>(sleepTime));
        }
    }

    CloseSock(sock);
    PlatformShutdown();
    return 0;
}

4. Client tối giản để test: client_main.cpp

Client gửi vị trí giả lập (di chuyển tròn đơn giản theo thời gian, không cần bàn phím/chuột thật) lên server 20 lần/giây, và in ra console mọi snapshot nhận được từ server.

C++ — client_main.cpp
// Bien dich Windows (MinGW-w64): g++ client_main.cpp -lws2_32 -o client.exe
// Bien dich Linux:                g++ client_main.cpp -o client
// Chay: client <dia_chi_ip_server>   (vi du: client 127.0.0.1)
#include "Protocol.h"
#include <chrono>
#include <thread>
#include <cmath>

constexpr uint16_t SERVER_PORT = 9000;
constexpr double TICK_INTERVAL = 1.0 / 20.0;

double NowSeconds() {
    using namespace std::chrono;
    return duration<double>(steady_clock::now().time_since_epoch()).count();
}

void SendInput(socket_t sock, const sockaddr_in& serverAddr, uint16_t seq, float x, float y) {
    uint8_t buf[64];
    PacketWriter w{ buf, sizeof(buf) };
    WriteHeader(w, PacketType::Input, seq, 8);
    w.WriteFloat(x);
    w.WriteFloat(y);
    sendto(sock, (const char*)buf, (int)w.offset, 0,
           (const sockaddr*)&serverAddr, sizeof(serverAddr));
}

void ProcessIncomingSnapshots(socket_t sock) {
    uint8_t buf[MAX_PACKET_SIZE];
    for (;;) {
        sockaddr_in from{};
#ifdef _WIN32
        int fromLen = sizeof(from);
#else
        socklen_t fromLen = sizeof(from);
#endif
        int n = recvfrom(sock, (char*)buf, sizeof(buf), 0, (sockaddr*)&from, &fromLen);
        if (n <= 0) { if (!WasWouldBlock()) {} break; }

        PacketReader r{ buf, (size_t)n };
        PacketHeader h;
        if (!ReadHeader(r, h) || h.type != PacketType::Snapshot) continue;

        uint8_t count;
        r.ReadU8(count);
        printf("Snapshot #%u: %u nguoi choi -> ", h.sequence, count);
        for (int i = 0; i < count; ++i) {
            uint16_t pid; float x, y;
            r.ReadU16(pid); r.ReadFloat(x); r.ReadFloat(y);
            printf("[id=%u x=%.2f y=%.2f] ", pid, x, y);
        }
        printf("\n");
    }
}

int main(int argc, char** argv) {
    const char* serverIp = (argc > 1) ? argv[1] : "127.0.0.1";
    PlatformStartup();

    socket_t sock = socket(AF_INET, SOCK_DGRAM, IPPROTO_UDP);
    SetSocketNonBlocking(sock);

    sockaddr_in serverAddr{};
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(SERVER_PORT);
    inet_pton(AF_INET, serverIp, &serverAddr.sin_addr);

    printf("Client dang gui du lieu gia lap toi server %s:%d\n", serverIp, SERVER_PORT);

    uint16_t seq = 0;
    double startTime = NowSeconds();

    while (true) {
        double frameStart = NowSeconds();
        double t = frameStart - startTime;

        // Gia lap di chuyen theo quy dao tron de de quan sat tren console
        float x = 100.0f * (float)cos(t);
        float y = 100.0f * (float)sin(t);
        SendInput(sock, serverAddr, seq++, x, y);

        ProcessIncomingSnapshots(sock);

        double elapsed = NowSeconds() - frameStart;
        double sleepTime = TICK_INTERVAL - elapsed;
        if (sleepTime > 0) {
            std::this_thread::sleep_for(std::chrono::duration<double>(sleepTime));
        }
    }

    CloseSock(sock);
    PlatformShutdown();
    return 0;
}
Cách thử nghiệm nhanh: Biên dịch cả hai file trên cùng máy (đặt Protocol.h cùng thư mục), chạy server ở một cửa sổ terminal, rồi chạy client 127.0.0.1 ở hai hoặc ba cửa sổ terminal khác — mỗi client sẽ thấy console in ra snapshot chứa vị trí của tất cả các client đang chạy, chuyển động theo quỹ đạo tròn lệch pha nhau tùy thời điểm khởi động.

5. Bài tập mở rộng cho người đọc

Ví dụ trên cố tình giữ tối giản để tập trung vào luồng dữ liệu cốt lõi. Trước khi bước sang Phần III/IV học cách các middleware thương mại (Photon, Unity Transport, FishNet) giải quyết chính những vấn đề này một cách hoàn thiện hơn, hãy tự thử mở rộng ví dụ theo các hướng sau để củng cố tư duy:

  • Thêm kênh Reliable cho chat: áp dụng cơ chế sequence/ACK/retransmission ở chương 11 để thêm một loại gói PacketType::Chat đảm bảo luôn tới nơi, chạy song song với kênh Unreliable hiện có cho vị trí.
  • Thêm interpolation phía client: thay vì vẽ ngay vị trí mới nhận được (gây giật do tick rate server chỉ 20Hz), lưu lại snapshot gần nhất và snapshot trước đó, nội suy (interpolate) mượt giữa hai điểm đó theo thời gian thực khi hiển thị — kỹ thuật đã học ở chương 6, Phần I.
  • Thêm Interest Management đơn giản: thay vì broadcast toàn bộ người chơi cho mọi client (như ví dụ này đang làm), chỉ gửi thông tin những người chơi trong một bán kính nhất định quanh mỗi client — áp dụng khái niệm đã học ở chương 7, Phần I, giúp giảm đáng kể băng thông khi số người chơi tăng lên hàng trăm/nghìn.
  • Tách luồng Network/Simulation: áp dụng kiến trúc hàng đợi đã học ở chương 13 thay vì chạy hoàn toàn tuần tự như ví dụ này.

Tổng kết Phần II

Sáu chương của Phần II đã đi từ ôn tập socket API chuyên biệt cho game (non-blocking, phân biệt client qua địa chỉ nguồn), qua thiết kế giao thức nhị phân an toàn portable, xây dựng lớp reliable UDP tự chế, cơ chế phát hiện mất kết nối, kiến trúc luồng đúng đắn, và cuối cùng ghép tất cả thành một game server UDP chạy được thực sự.
